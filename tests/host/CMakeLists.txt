cmake_minimum_required(VERSION 3.20)
project(tempest_regressions C)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(MAIN ${CMAKE_CURRENT_LIST_DIR}/../../firmware/main)
add_executable(weather_tests test_weather.c test_runtime.c platform.c
    ${MAIN}/history.c ${MAIN}/wx_state.c ${MAIN}/wx_daily.c ${MAIN}/alert_policy.c ${MAIN}/tempest_ws.c ${MAIN}/ota_health.c)
target_include_directories(weather_tests PRIVATE stubs ${MAIN})
target_compile_definitions(weather_tests PRIVATE CONFIG_TEMPEST_OBS_STALE_S=180 CONFIG_INDOOR_STALE_S=1800)
if(MSVC)
    target_compile_options(weather_tests PRIVATE /FI${CMAKE_CURRENT_LIST_DIR}/stubs/test_platform.h /UNDEBUG)
    target_compile_definitions(weather_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(weather_tests PRIVATE -include ${CMAKE_CURRENT_LIST_DIR}/stubs/test_platform.h -UNDEBUG)
    target_link_libraries(weather_tests PRIVATE m)
endif()
add_test(NAME weather_regressions COMMAND weather_tests)
